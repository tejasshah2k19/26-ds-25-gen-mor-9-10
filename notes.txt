boolean findTarget(int targetSum){ 
    int low=0;
    int high = SIZE-1 ;  
    
    while(low<=high){
        int sum  = arr[low] + arr[high];    
        if( sum == targetSum ){
            return true; 
        }else if( sum > targetSum){
            high--; 
        }else{
            low++; 
        }
    }
    return false; 
}
